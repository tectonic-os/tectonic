### Base Image
FROM ${BASE} AS rootfs

## Module layers

# ---- /opt build root ----
RUN mv /opt /opt.bak && mkdir /opt

# ---- core/first ----
RUN --mount=type=bind,from=ctx,source=/modules/core/first,target=/ctx/modules/core/first \
    --mount=type=bind,from=ctx,source=/generated/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/broken-overlay/modules/core/first.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- core/second ----
RUN --mount=type=bind,from=ctx,source=/modules/core/second,target=/ctx/modules/core/second \
    --mount=type=bind,from=ctx,source=/generated/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/broken-overlay/modules/core/second.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \

    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- core/third ----
RUN --mount=type=bind,from=ctx,source=/modules/core/third,target=/ctx/modules/core/third \

    --mount=type=bind,from=ctx,source=/generated/broken-overlay/modules/core/third.sh,target=/ctx/module.sh \

# ---- flavour gate ----
ARG FLAVOUR

# ---- image version ----
ARG IMAGE_VERSION=dev

# ---- base family ----
ARG FAMILY

# ---- declared conformance ----
ARG CONFORMS=

# ---- claimed and refused rules ----
ARG SCAP_CONTENT=
ARG SCAP_CLAIMED=
ARG SCAP_REFUSED=

# ---- image identity ----
ARG IMAGE_ID="broken-overlay"
ARG IMAGE_NAME="Broken Overlay"
ARG IMAGE_REGISTRY=""

# ---- os-release ----
RUN --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    IMAGE_VERSION="${IMAGE_VERSION}" IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" /ctx/tect os-release

# ---- finalize ----
RUN --mount=type=bind,from=ctx,source=/generated/broken-overlay/finalize.sh,target=/ctx/finalize.sh \
    --mount=type=bind,from=ctx,source=/generated/lib,target=/ctx/lib \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    FLAVOUR=${FLAVOUR} IMAGE_VERSION="${IMAGE_VERSION}" \
    CONFORMS="${CONFORMS}" FAMILY="${FAMILY}" \
    SCAP_CONTENT="${SCAP_CONTENT}" SCAP_CLAIMED="${SCAP_CLAIMED}" \
    SCAP_REFUSED="${SCAP_REFUSED}" \
    IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" bash /ctx/finalize.sh
