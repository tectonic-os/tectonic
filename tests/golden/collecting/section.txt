### Base Image
FROM quay.io/fedora/fedora-bootc:44

## Build phases and modules

# ---- core/menu ----
RUN --mount=type=bind,from=ctx,source=/modules/core/menu,target=/ctx/modules/core/menu \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/lib/run-module.sh /ctx/modules/core/menu

# ---- core/power ----
RUN --mount=type=bind,from=ctx,source=/modules/core/power,target=/ctx/modules/core/power \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    MODULE_COLLECT="justfile.inc=/usr/share/menu/justfile.apps.d/0900-core-power.part" bash /ctx/lib/run-module.sh /ctx/modules/core/power

# ---- core/network ----
RUN --mount=type=bind,from=ctx,source=/modules/core/network,target=/ctx/modules/core/network \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    MODULE_COLLECT="justfile.inc=/usr/share/menu/justfile.apps.d/0100-core-network.part" bash /ctx/lib/run-module.sh /ctx/modules/core/network

# ---- flavour gate ----
ARG FLAVOUR

# ---- finalize hook order ----
ARG FINALIZE_ORDER=""

# ---- collected file destinations ----
ARG COLLECT_TARGETS="/usr/share/menu/justfile.apps"

# ---- image version ----
ARG IMAGE_VERSION=dev

# ---- image identity ----
ARG IMAGE_ID="collecting"
ARG IMAGE_NAME="Collecting"
ARG IMAGE_REGISTRY=""
