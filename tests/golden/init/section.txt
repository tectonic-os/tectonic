### Base Image
FROM ${BASE}

## Module layers

# ---- flavour gate ----
ARG FLAVOUR

# ---- image version ----
ARG IMAGE_VERSION=dev

# ---- image identity ----
ARG IMAGE_ID="example"
ARG IMAGE_NAME="Example"
ARG IMAGE_URL="https://github.com/someone/example"
ARG IMAGE_ISSUES_URL="https://github.com/someone/example/issues"
ARG IMAGE_REGISTRY=""

# ---- os-release ----
RUN --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    IMAGE_VERSION="${IMAGE_VERSION}" IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_URL="${IMAGE_URL}" IMAGE_ISSUES_URL="${IMAGE_ISSUES_URL}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" /ctx/tect os-release
