### Base Image
FROM ${BASE}

## Module layers

# ---- /opt build root ----
RUN mv /opt /opt.bak && mkdir /opt

# ---- fedora-family ----
RUN --mount=type=bind,from=ctx,source=/modules/fedora-family,target=/ctx/modules/fedora-family \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib,rw \
    --mount=type=bind,from=ctx,source=/modules/fedora-family/lib/family.sh,target=/ctx/lib/family.sh \
    --mount=type=bind,from=ctx,source=/generated/example.d/fedora-family.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- image version ----
ARG IMAGE_VERSION=dev

# ---- base family ----
ARG FAMILY

# ---- image identity ----
ARG IMAGE_ID="example"
ARG IMAGE_NAME="Example"
ARG IMAGE_URL="https://github.com/someone/example"
ARG IMAGE_ISSUES_URL="https://github.com/someone/example/issues"
ARG IMAGE_REGISTRY=""

# ---- os-release ----
RUN --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    IMAGE_VERSION="${IMAGE_VERSION}" IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_URL="${IMAGE_URL}" IMAGE_ISSUES_URL="${IMAGE_ISSUES_URL}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" /ctx/tect os-release

# ---- finalize ----
RUN --mount=type=bind,from=ctx,source=/generated/example.d/finalize.sh,target=/ctx/finalize.sh \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    FLAVOUR=${FLAVOUR} IMAGE_VERSION="${IMAGE_VERSION}" \
    IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_URL="${IMAGE_URL}" IMAGE_ISSUES_URL="${IMAGE_ISSUES_URL}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" bash /ctx/finalize.sh
