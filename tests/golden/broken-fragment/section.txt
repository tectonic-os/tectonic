### Base Image
FROM quay.io/fedora/fedora-bootc:44

## Build phases and modules

# ---- phase 00-setup.sh ----
RUN --mount=type=bind,from=ctx,source=/00-setup.sh,target=/ctx/00-setup.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    /ctx/00-setup.sh

# ---- core/early ----
# verbatim from modules/core/early/Containerfile.inc:
RUN echo "building ${FLAVOUR}" > /usr/share/early

RUN --mount=type=bind,from=ctx,source=/modules/core/early,target=/ctx/modules/core/early \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/broken-fragment.d/core/early.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- flavour gate ----
ARG FLAVOUR

# ---- [dev] ----
# ---- core/silent ----
RUN --mount=type=bind,from=ctx,source=/modules/core/silent,target=/ctx/modules/core/silent \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/broken-fragment.d/core/silent@dev.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# verbatim from modules/core/silent/Containerfile.inc:
RUN echo silent > /usr/share/silent

# ---- [dev] ----
# ---- core/mismatched ----
# verbatim from modules/core/mismatched/Containerfile.inc:
RUN FLAVOUR_GATE=server echo mismatched > /usr/share/mismatched

RUN --mount=type=bind,from=ctx,source=/modules/core/mismatched,target=/ctx/modules/core/mismatched \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/broken-fragment.d/core/mismatched@dev.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- finalize hook order ----
ARG FINALIZE_ORDER=""

# ---- collected file destinations ----
ARG COLLECT_TARGETS=""

# ---- image version ----
ARG IMAGE_VERSION=dev

# ---- image identity ----
ARG IMAGE_ID="broken-fragment"
ARG IMAGE_NAME="Broken Fragment"
ARG IMAGE_REGISTRY=""

# ---- phase 99-finalize.sh ----
RUN --mount=type=bind,from=ctx,source=/99-finalize.sh,target=/ctx/99-finalize.sh \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/modules,target=/ctx/modules \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    FLAVOUR=${FLAVOUR} IMAGE_VERSION=${IMAGE_VERSION} FINALIZE_ORDER="${FINALIZE_ORDER}" \
    COLLECT_TARGETS="${COLLECT_TARGETS}" \
    IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" /ctx/99-finalize.sh
