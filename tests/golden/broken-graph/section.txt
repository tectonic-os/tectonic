### Base Image
FROM ${BASE}

## Module layers

# ---- /opt build root ----
RUN mv /opt /opt.bak && mkdir /opt

# ---- core/one ----
RUN --mount=type=bind,from=ctx,source=/modules/core/one,target=/ctx/modules/core/one \
    --mount=type=bind,from=ctx,source=/generated/lib,target=/ctx/lib,rw \
    --mount=type=bind,from=ctx,source=/modules/core/one/lib/one/shared.sh,target=/ctx/lib/shared.sh \
    --mount=type=bind,from=ctx,source=/modules/core/two/lib/two/shared.sh,target=/ctx/lib/shared.sh \
    --mount=type=bind,from=ctx,source=/generated/broken-graph/modules/core/one.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- core/two ----
RUN --mount=type=bind,from=ctx,source=/modules/core/two,target=/ctx/modules/core/two \
    --mount=type=bind,from=ctx,source=/generated/lib,target=/ctx/lib,rw \
    --mount=type=bind,from=ctx,source=/modules/core/one/lib/one/shared.sh,target=/ctx/lib/shared.sh \
    --mount=type=bind,from=ctx,source=/modules/core/two/lib/two/shared.sh,target=/ctx/lib/shared.sh \
    --mount=type=bind,from=ctx,source=/generated/broken-graph/modules/core/two.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \

# ---- core/three ----
RUN --mount=type=bind,from=ctx,source=/modules/core/three,target=/ctx/modules/core/three \
    --mount=type=bind,from=ctx,source=/generated/lib,target=/ctx/lib,rw \
    --mount=type=bind,from=ctx,source=/modules/core/one/lib/one/shared.sh,target=/ctx/lib/shared.sh \
    --mount=type=bind,from=ctx,source=/modules/core/two/lib/two/shared.sh,target=/ctx/lib/shared.sh \

# ---- flavour gate ----
ARG FLAVOUR

# ---- image version ----
ARG IMAGE_VERSION=dev

# ---- base family ----
ARG FAMILY

# ---- image identity ----
ARG IMAGE_ID="broken-graph"
ARG IMAGE_NAME="Broken Graph"
ARG IMAGE_REGISTRY=""

# ---- os-release ----
RUN --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    IMAGE_VERSION="${IMAGE_VERSION}" IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" /ctx/tect os-release

# ---- finalize ----
RUN --mount=type=bind,from=ctx,source=/generated/broken-graph/finalize.sh,target=/ctx/finalize.sh \
    --mount=type=bind,from=ctx,source=/generated/lib,target=/ctx/lib \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    FLAVOUR=${FLAVOUR} IMAGE_VERSION="${IMAGE_VERSION}" \
    IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" bash /ctx/finalize.sh
