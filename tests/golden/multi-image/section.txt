### Base Image
FROM ${BASE}

## Module layers

# ---- core/shared ----
RUN --mount=type=bind,from=ctx,source=/modules/core/shared,target=/ctx/modules/core/shared \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/workstation.d/core/shared.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- flavour gate ----
ARG FLAVOUR

# ---- [dev] ----
# ---- core/dev-tools ----
RUN --mount=type=bind,from=ctx,source=/modules/core/dev-tools,target=/ctx/modules/core/dev-tools \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/workstation.d/core/dev-tools@dev.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- image version ----
ARG IMAGE_VERSION=dev

# ---- base family ----
ARG FAMILY

# ---- image identity ----
ARG IMAGE_ID="workstation"
ARG IMAGE_NAME="Workstation"
ARG IMAGE_REGISTRY=""

# ---- os-release ----
RUN --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    IMAGE_VERSION="${IMAGE_VERSION}" IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" /ctx/tect os-release

# ---- finalize ----
RUN --mount=type=bind,from=ctx,source=/generated/workstation.d/finalize.sh,target=/ctx/finalize.sh \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/modules/core/shared,target=/ctx/modules/core/shared \
    --mount=type=bind,from=ctx,source=/modules/core/dev-tools,target=/ctx/modules/core/dev-tools \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    FLAVOUR=${FLAVOUR} IMAGE_VERSION="${IMAGE_VERSION}" \
    IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" bash /ctx/finalize.sh
