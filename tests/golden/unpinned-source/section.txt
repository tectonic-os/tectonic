### Base Image
FROM ${BASE}

## Module layers

# ---- tectonic-os/hello ----
RUN --mount=type=bind,from=ctx,source=/modules/.remote/tectonic-os/hello,target=/ctx/modules/.remote/tectonic-os/hello \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/example.d/tectonic-os/hello.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- flavour gate ----
ARG FLAVOUR

ARG IMAGE_VERSION=dev

ARG FAMILY
