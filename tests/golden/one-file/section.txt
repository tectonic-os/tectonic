### Base Image
FROM ${BASE} AS rootfs

## Module layers

# ---- /opt build root ----
RUN mv /opt /opt.bak && mkdir /opt

# ---- fedora-family ----
RUN --mount=type=bind,from=ctx,source=/modules/fedora-family,target=/ctx/modules/fedora-family \
    --mount=type=bind,from=ctx,source=/generated/lib,target=/ctx/lib,rw \
    --mount=type=bind,from=ctx,source=/modules/fedora-family/lib/family.sh,target=/ctx/lib/family.sh \
    --mount=type=bind,from=ctx,source=/generated/alpha/modules/fedora-family.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- core/hello ----
RUN --mount=type=bind,from=ctx,source=/modules/core/hello,target=/ctx/modules/core/hello \
    --mount=type=bind,from=ctx,source=/generated/lib,target=/ctx/lib,rw \
    --mount=type=bind,from=ctx,source=/modules/fedora-family/lib/family.sh,target=/ctx/lib/family.sh \
    --mount=type=bind,from=ctx,source=/generated/alpha/modules/core/hello.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- flavour gate ----
ARG FLAVOUR

# ---- image version ----
ARG IMAGE_VERSION=dev

# ---- base family ----
ARG FAMILY

# ---- image identity ----
ARG IMAGE_ID="alpha"
ARG IMAGE_NAME="Alpha"
ARG IMAGE_REGISTRY=""

# ---- os-release ----
RUN --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    IMAGE_VERSION="${IMAGE_VERSION}" IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" /ctx/tect os-release

# ---- finalize ----
RUN --mount=type=bind,from=ctx,source=/generated/alpha/finalize.sh,target=/ctx/finalize.sh \
    --mount=type=bind,from=ctx,source=/generated/lib,target=/ctx/lib \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    FLAVOUR=${FLAVOUR} IMAGE_VERSION="${IMAGE_VERSION}" \
    IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" bash /ctx/finalize.sh

# verbatim from modules/core/hello/Containerfile.inc:
# The lineage stage has ended by here, so it can be bound from a stage of its
# own. Whatever a tail fragment opens last is the final image.
FROM rootfs AS greeting
RUN --mount=type=bind,from=rootfs,source=/,target=/rootfs,rw \
    cp /rootfs/etc/os-release /os-release

FROM rootfs
COPY --from=greeting /os-release /usr/share/hello-os-release
