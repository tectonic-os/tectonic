### Base Image
FROM ${BASE}

## Module layers

# ---- core/a ----
RUN --mount=type=bind,from=ctx,source=/modules/core/a,target=/ctx/modules/core/a \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/broken-cycle.d/core/a.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- core/b ----
RUN --mount=type=bind,from=ctx,source=/modules/core/b,target=/ctx/modules/core/b \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/broken-cycle.d/core/b.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- flavour gate ----
ARG FLAVOUR

# ---- image version ----
ARG IMAGE_VERSION=dev

# ---- base family ----
ARG FAMILY

# ---- image identity ----
ARG IMAGE_ID="broken-cycle"
ARG IMAGE_NAME="Broken Cycle"
ARG IMAGE_REGISTRY=""

# ---- os-release ----
RUN --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    IMAGE_VERSION="${IMAGE_VERSION}" IMAGE_ID="${IMAGE_ID}" IMAGE_NAME="${IMAGE_NAME}" IMAGE_REGISTRY="${IMAGE_REGISTRY}" /ctx/tect os-release
