### Base Image
FROM quay.io/fedora/fedora-bootc:44

## Build phases and modules

# ---- core/hello ----
RUN --mount=type=bind,from=ctx,source=/modules/core/hello,target=/ctx/modules/core/hello \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/pinned.d/core/hello.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh

# ---- steam-tweaks ----
RUN --mount=type=bind,from=ctx,source=/modules/.remote/steam-tweaks,target=/ctx/modules/.remote/steam-tweaks \
    --mount=type=bind,from=ctx,source=/lib,target=/ctx/lib \
    --mount=type=bind,from=ctx,source=/generated/pinned.d/steam-tweaks.sh,target=/ctx/module.sh \
    --mount=type=bind,from=tect,source=/tect,target=/ctx/tect \
    --mount=type=cache,target=/var/cache \
    --mount=type=cache,target=/var/log \
    --mount=type=tmpfs,target=/tmp \
    bash /ctx/module.sh
